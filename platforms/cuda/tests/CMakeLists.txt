#
# Testing
#

# The InteraxFF Reference plugin (the reference of some tests) and the AMOEBA plugins of OpenMM (HippoNonbondedForce)
# are loaded at run time.
SET(OPENMM_AMOEBA_REFERENCE_PLUGIN "${OPENMM_DIR}/lib/plugins/${CMAKE_SHARED_LIBRARY_PREFIX}OpenMMAmoebaReference${CMAKE_SHARED_LIBRARY_SUFFIX}")
SET(OPENMM_AMOEBA_CUDA_PLUGIN "${OPENMM_DIR}/lib/plugins/${CMAKE_SHARED_LIBRARY_PREFIX}OpenMMAmoebaCUDA${CMAKE_SHARED_LIBRARY_SUFFIX}")

# Automatically create tests using files named "Test*.cpp", run in each precision as OpenMM's CUDA tests are.
FILE(GLOB TEST_PROGS "*Test*.cpp")
FOREACH(TEST_PROG ${TEST_PROGS})
    GET_FILENAME_COMPONENT(TEST_ROOT ${TEST_PROG} NAME_WE)
    ADD_EXECUTABLE(${TEST_ROOT} ${TEST_PROG})
    TARGET_LINK_LIBRARIES(${TEST_ROOT} ${SHARED_TARGET} OpenMMAmoeba)
    ADD_DEPENDENCIES(${TEST_ROOT} InteraxPluginReference)
    TARGET_COMPILE_DEFINITIONS(${TEST_ROOT} PRIVATE OPENMM_AMOEBA_REFERENCE_PLUGIN="${OPENMM_AMOEBA_REFERENCE_PLUGIN}"
        OPENMM_AMOEBA_CUDA_PLUGIN="${OPENMM_AMOEBA_CUDA_PLUGIN}" INTERAX_REFERENCE_PLUGIN="$<TARGET_FILE:InteraxPluginReference>")
    SET_TARGET_PROPERTIES(${TEST_ROOT} PROPERTIES LINK_FLAGS "${EXTRA_COMPILE_FLAGS}" COMPILE_FLAGS "${EXTRA_COMPILE_FLAGS}")
    ADD_TEST(${TEST_ROOT}Single ${EXECUTABLE_OUTPUT_PATH}/${TEST_ROOT} single)
    ADD_TEST(${TEST_ROOT}Mixed ${EXECUTABLE_OUTPUT_PATH}/${TEST_ROOT} mixed)
    ADD_TEST(${TEST_ROOT}Double ${EXECUTABLE_OUTPUT_PATH}/${TEST_ROOT} double)
ENDFOREACH(TEST_PROG ${TEST_PROGS})
